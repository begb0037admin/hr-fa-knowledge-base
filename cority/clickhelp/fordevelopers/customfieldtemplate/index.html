Example: Get custom field template names and custom fields

select

t.name as CustomFieldTemplate,

t.CustomField.Name as CustomField

from CustomFieldTemplate t

order by t.name

Example: Get custom fields for a POI, then get custom field values.

select

p.name,

p.CustomFieldTemplate.Name,

p.CustomFieldTemplate.CustomField.Name

from POI p

where p.name like 'Criteria Pollutants'

 

-- then get custom field values

select

p.name,

p."Construction Permit",

p."Date Constructed",

p."Emission Control Equipment",

p."Firing Rate (mmBtu/hr)",

p."Permit Emission Unit Number",

p."Permit Unit Number"

from POI p

where p.name like 'Criteria Pollutants'

Example: Get custom field names and order in template.

select

cf.name as TemplateName,

cf.CustomField.Name as CustomFieldName,

cf.CustomField.[Order]

from CustomFieldTemplate cf

Permissions:  Manage Custom Fields right is required to query this table.

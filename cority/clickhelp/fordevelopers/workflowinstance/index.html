Example: Get current step of a Workflow.

select

step.ID,

step.Name,

step.CompletedDate,

step.CompletedBy,

wi.Name as WorkflowName

from WorkflowInstance wi

join WorkflowStep step on wi

where wi.name like '%Workflow%'

Example: Get a count of workflow instances where the value of a form field matches a string.

select

count(distinct wi.UniqueID) as NumberOfWorkflows,

wi.UniqueID

from WorkflowInstance wi

join WorkflowType wt on wi

join WorkflowStep ws on wi

where wt.name like 'list box filter test'

and ws."List box" = 'One Value'

--and ws."List box with one value" is null

group by wi.UniqueID

Example: Get task and associated workflow instances for specified date range

select

t.ID as TaskID,

t.Name As TaskName,

ti.DueDateTime As TaskInstanceDueDate,

t.VersionID As TaskVersionID,

co.Name As ObjectName,

wi.Name As WFName,

wi.ID

from Task t

join TaskInstance ti on t

join ComplianceObject co on co.ID = ti.ObjectID

join WorkflowInstance wi on wi.Task.InstanceDueDateTime = ti.DueDateTime

Where t.Object.Path Like '%parent object path%'

and ti.DueDateTime >= '2017-01-30' and ti.DueDateTime <= '2020-12-31'

and ti.DueDateTime = wi.Task.InstanceDueDateTime

and ti.TaskVersionID = wi.Task.VersionID

order by ti.DueDateTime

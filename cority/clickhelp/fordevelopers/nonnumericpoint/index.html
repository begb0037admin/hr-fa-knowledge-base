Example: Get non-numeric data (task data) for a non-numeric requirement.

select

r.Name,

np.Collector,

np.Complete,

np.BeginDate,

np.EndDate,

np.Comply

from NonNumericPoint np

join Requirement r on np

where

r.Name like 'Breathing Air Laboratory Analysis'

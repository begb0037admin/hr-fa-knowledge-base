Example: Get requirement data points for POI with name string containing "outfall".

select

r.name as Requirement,

p.name as POI,

f.name as Facility,

d.Value,

d.Collected

from DataPoint d

join Requirement r on d

join POI p on r is under p

join Facility f on p is under f

where p.name like '%outfall%'

Example: List the requirements in the system and count the number of requirements with the same name.

select

r.Name,

count(r.Name) as RequirementCount

from

Requirement r

group by

r.Name

order by

r.Name

Example: Get Time-Based Fixed Calculated Requirements, with time zone, UOM and calculation script, and datapoint value for 12/31/2013.

select

r.name,

r.Type,

r.TimeZone,

r.CalculationUOM,

rs.ScriptWithFullPath,

rs.BeginDate,

d.Value

from Requirement r

join RequirementScript rs on r

join DataPoint d on r

join Facility f on f is above r

join TimeZone tz on f

where localtime(d.Collected,tz) = '12/31/2013'

and r.Type like 'Time-Based Calculated (Fixed period) Requirement'

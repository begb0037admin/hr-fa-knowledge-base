Example 1: List POIs aggregating by UDF "udfNumeric" where child event instance dates are between 01/01/2012 and 10/20/2012.

select

p.Name,

sum(ei."udfNumeric") as total

from

Facility f

join POI p on p is under f

join EventLog el on el is under p

join EventInstance ei on el

where

ei.BeginDate > '01/01/2012' and

ei.EndDate < '10/20/2012'

group by

p.Name

Example 2: Get the names and tags for event instances in a given event log and Division (division name is the state).

select top 10

d.Name,

el.Name as Log,

el."Unique Tag-Integration" as elTag,

ei.Name as Instance,

ei."Unique Tag-Integration" as eiTag,

ei."Manifest Number"

from

EventInstance ei

join EventLog el on ei

join Division d on d is above el

where

el.Name like 'Used Tire%'

and

d.Name = 'Alaska'

order by

d.Name, el.Name, ei.Name
